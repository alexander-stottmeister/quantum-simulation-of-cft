% Supp Fig. 7 (fig:momrgerrorHS): momentum-cutoff RG, off-diagonal Hilbert-Schmidt error,
% Neveu-Schwarz sector, M = 3 (the bound carries no chi_{Gamma_M}), k = +-2, +-3, +-4.
% The quantity is even in k and vanishes identically for k = 0, +-1 (Moebius generators).
% Documented in ../FIGURES.md.

  ser/.style={line width=.5pt, mark size=1.6pt, mark options={fill opacity=.85}},

\begin{document}
\begin{tikzpicture}
\begin{axis}[
  % --- geometry and scales ---
  width=308pt, height=186pt, scale only axis,
  ymode=log, log basis y=10,
  % --- labels ---

  xlabel style={font=\footnotesize}, ylabel style={font=\footnotesize},
  tick label style={font=\footnotesize},
